# 1. ElasticSearch = Elasticsearch is a distributed search and analytics engine built on Apache Lucene.
It’s designed to store, search, and analyze huge amounts of data in near real-time.Generally used as a secondary database specifically for the search,filteration,sorting or read heavy tasks.Never used elastic search as your primary backend and store all the priamry data init.All data store in the primary backend only the data that are read heavy are stored in the elastic search.It integreated with any database but make sure one think firstly data store,update,delete in the primary after that in elastic.


# Elastic generally contains = indexes,document.
# where , indexed = means the table or database.
# document = means the json document inside it.which contain _id,_source._id is unique and _source contains the raw json data.

# Used to fetch all the available indexes:
GET /_cat/indices?v


# Used to fetch all the document present in a single index:
GET /my-index/_search
{   
    "size":20, 
    "query": {
        "match_all": {}
    }
}

# Used to fetch any specific document from the index:
GET /my-index/_doc/qLA6UZsBWKXM1izZyzIQ

# 2. Index Settings:

# create new index syntax = PUT /new_idx_name

# Check if already exists or not = return 200 if yes else 404:
HEAD /my-index

# Get index mapping + setings:
GET /my-index

# Get only settings:
GET /my-index/_settings

# Update index settings = Curretly unavalible in serverless mode
PUT /my-index/_settings
{
  "index": {
    "number_of_replicas": 1
  }
}

# Used to delete index permanently:
DELETE /my-index

# 3. Mapping of the indexes = It is similar to the schema designing in other database.It tells the elastic which field store what type of data.If you not specify mapping before adding the document then elastic automatically identifies it, which is not recommended.

# Check the mapping of the existing index:
GET /products/_mapping

# Create the schema of the one index:
PUT /mapped_products
{
  "mappings": {
    "properties": {
      "name": { 
        "type": "text",
        "fields":{
            "raw":{"type":"keyword"}
        }
        },
      "brand": { "type": "keyword" },
      "price": { "type": "integer" },
      "inStock": { "type": "boolean" }
    }
  }
}

# Here we you clearly see that even if we don't follow the schema/mapping but elastic does not gave any error.Elasticsearch is very flexible so it does not gave any error even if the mapping are wrong.

put /mapped_products/_doc/1
{
    "name":"Nike White show.",
    "brand":"Nike",
    "price":"999",
    "inStock":true
}

put /mapped_products/_doc/2
{
    "name":"Nike White show.",
    "brand":"Nike",
    "price":999
}

GET /mapped_products/_search
{
    "query": {
        "match_all":{}
    }
}

# Mapping data types:

PUT /dummy_mapping_Example
{
  "mappings": {
    "properties": {
      "name": { "type": "text" },
      "brand": { "type": "keyword" },
        # Elastic automatically create an array.
      "tags": {
        "type": "keyword"
      },
        # properties field used for object
      "seller": {
        "properties": {
          "id": { "type": "keyword" },
          "name": { "type": "text" }
        }
      },
        # nestd type used in case of array of objects.
      "variants": {
        "type": "nested",
        "properties": {
          "color": { "type": "keyword" },
          "size": { "type": "keyword" },
          "price": { "type": "integer/float/double" }
        }
      }
    }
  }
}

# 4. CRUD Operation= The elastic search crud operations are little bit different from the other.Here we use mostly get,put,post,delete.Put used to create new index and add document with unique id and replace specific document data completely with unique id.Here post used for add document but unique id generated by the elastic itself and post with _update used to update the specfic document with specific field , rest unchanged.
GET /crud_index/_search
{
    "query":{
        "match_all":{}
    }
}
# Create index:
PUT /crud_index

# Add document or overrite:
PUT /crud_index/_doc/1
{
    "name":"tmep1",
    "class":"10"
}

# Now in order to add the multiple bulk new data we used the post because post is create to do bulk task.Put is only used to do single idempotent jobs.

POST /_bulk
{ "index": { "_index": "crud_index", "_id": "2" } }{ "name": "temp2", "class": "9" }{ "index": { "_index": "crud_index", "_id": "3" } }{ "name": "temp3", "class": "11" }

# Single document updation:

POST /crud_index/_update/2
{
    "doc":{
        "class":10
    }
}

# Multiple documents updation:
POST /_bulk
{ "update": { "_index": "crud_index", "_id": "1" } }{ "doc": { "class": "12" } }{ "update": { "_index": "crud_index", "_id": "2" } }{ "doc": { "class": "10" } }{ "update": { "_index": "crud_index", "_id": "3" } }{ "doc": { "class": "11" } }

# or 

# Generally used for conditional ulk updates:
POST /crud_index/_update_by_query
{
  "script": {
    "source": "ctx._source.class = '11'"
  },
  "query": {
    "term": {
      "class": "10"
    }
  }
}

# Used to get bulk data at ones:
GET /crud_index/_mget
{
  "ids": ["1", "2", "3"]
}

# Note = 1. put = used to create new index and single document insertion.
# 2. post = used for the bulk insertion,updation,filteration,queries,search,match,etc.
